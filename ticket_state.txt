<script>
    let allTickets = $("input[name^='ticket']");

    $('.btn-number').click(function(e){
        e.preventDefault();

        fieldName = $(this).attr('data-field');
        type      = $(this).attr('data-type');
        var input = $("input[name='"+fieldName+"']");
        var currentVal = parseInt(input.val());
        if (!isNaN(currentVal)) {
            if(type === 'minus') {

                if(currentVal > input.attr('min')) {
                    input.val(currentVal - 1).change();
                }
                if(parseInt(input.val()) == input.attr('min')) {
                    $(this).attr('disabled', true);
                    $(this).removeClass('text-danger')
                }

            } else if(type === 'plus') {
                if(currentVal < input.attr('max')) {
                    if (currentVal < input.attr('data-max')) {
                        input.val(currentVal + 1).change();
                    }
                    if (currentVal == (input.attr('data-max') - 1)) {
                        $(this).attr('disabled', true);
                        $(this).removeClass('text-success');
                    }
                }
            }
        } else {
            input.val(0);
        }
    });

    let inputNumber = $('.input-number');

    inputNumber.focusin(function(){
        $(this).data('oldValue', $(this).val());
    });

    inputNumber.change(function() {
        minValue =  parseInt($(this).attr('min'));
        maxValue =  parseInt($(this).attr('max'));
        valueCurrent = parseInt($(this).val());

        name = $(this).attr('name');
        if(valueCurrent >= minValue) {
            let minusButton = $(".btn-number[data-type='minus'][data-field='"+name+"']");
            minusButton.removeAttr('disabled')
            minusButton.addClass('text-danger')
        } else {
            $(this).val($(this).data('oldValue'));
        }

        if(isMaximum(maxValue + 1, $(this).attr('data-mid'))) {
            $(this).val($(this).data('oldValue'));
        }

        setPlus(isMaximum(maxValue, $(this).attr('data-mid')), $(this).attr('data-mid'))
    });

    inputNumber.prop('readonly', true)

    function isMaximum(maximum, mid) {
        let tickets = 0
        Array.from(allTickets).forEach(function(item) {
            if (item.getAttribute('data-mid') === mid) {
                tickets += parseInt(item.value);
            }
        })
        console.log(tickets)
        return tickets >= maximum
    }

    function setPlus(disabled, mid) {
        $('.btn-number').each(function () {
            if (($(this).attr('data-type') === 'plus') && $(this).attr('data-mid') === mid) {
                if (disabled) {
                    $(this).attr('disabled', true);
                    $(this).removeClass('text-success');
                } else {
                    $(this).removeAttr('disabled');
                    $(this).addClass('text-success')
                }
            }
        })
    }

</script>
